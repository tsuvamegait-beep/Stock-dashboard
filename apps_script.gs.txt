function out_(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}
function load_() {
  const p = PropertiesService.getScriptProperties();
  const n = +p.getProperty('n') || 0;
  if (!n) return {updatedAt: 0, data: null};
  let s = '';
  for (let i = 0; i < n; i++) s += p.getProperty('c' + i) || '';
  return JSON.parse(s);
}
function save_(obj) {
  const p = PropertiesService.getScriptProperties();
  const s = JSON.stringify(obj), n = Math.ceil(s.length / CHUNK), old = +p.getProperty('n') || 0;
  const all = {};
  for (let i = 0; i < n; i++) all['c' + i] = s.substr(i * CHUNK, CHUNK);
  all.n = String(n);
  p.setProperties(all);
  for (let i = n; i < old; i++) p.deleteProperty('c' + i);
}
function doGet(e) {
  if (!e || !e.parameter || e.parameter.key !== SECRET_KEY) return out_({ok: false, error: '비밀번호가 맞지 않습니다'});
  if (e.parameter.history) return out_({ok: true, history: readHistory_()});
  if (e.parameter.trades) return out_({ok: true, rows: readTrades_()});
  if (e.parameter.series) return out_(readSeries_(String(e.parameter.t || ''), String(e.parameter.from || '')));
  if (e.parameter.credit) return out_(e.parameter.refresh ? refreshCredit_() : readCredit_());
  if (e.parameter.peers) return out_(readPeers_(String(e.parameter.t || '')));
  if (e.parameter.fx) { const f = readFx_(); return f ? out_({ok: true, fx: f}) : out_({ok: false, error: '환율 탭이 없습니다. Apps Script에서 setupFx를 한 번 실행하세요'}); }
  const d = load_();
  return out_({ok: true, updatedAt: d.updatedAt, data: d.data});
}
function doPost(e) {
  let body;
  try { body = JSON.parse(e.postData.contents); } catch (err) { return out_({ok: false, error: '요청 형식 오류'}); }
  if (body.key !== SECRET_KEY) return out_({ok: false, error: '비밀번호가 맞지 않습니다'});
  if (body.action === 'saveTrades') return out_(saveTrades_(body.account, body.rows));
  if (body.action === 'saveCredit') return out_(saveCredit_(body.rows));
  if (body.action === 'deleteTrades') return out_(saveTrades_(body.account, []));
  const cur = load_();
  if (body.updatedAt < cur.updatedAt) return out_({ok: false, error: '서버에 더 최신 데이터가 있습니다. 새로고침 후 다시 시도하세요'});
  save_({updatedAt: body.updatedAt, data: body.data});
  return out_({ok: true, updatedAt: body.updatedAt});
}

/* ===== 매일 정해진 시각에 총자산 기록 ===== */
const norm_ = s => String(s == null ? '' : s).replace(/\s+/g, '').toLowerCase();
const num_ = s => { const m = String(s == null ? '' : s).replace(/,/g, '').match(/-?\d+(\.\d+)?/); return m ? parseFloat(m[0]) : NaN; };

// 시트 값(rows)과 대시보드가 올려둔 보유 현황(snap)으로 하루치 기록을 계산한다.
function calc_(snap, rows) {
  let hi = rows.findIndex(r => r.some(c => /^(현재가|price)$/i.test(String(c).trim())));
  if (hi < 0) throw new Error('시트에서 "현재가" 제목을 찾지 못했습니다');
  const H = rows[hi].map(c => String(c).trim().toLowerCase());
  const pc = H.findIndex(c => c === '현재가' || c === 'price'), lc = H.findIndex(c => c === '현지가');
  const codeCol = H.findIndex(c => ['종목코드', '티커', 'ticker', 'code', 'symbol'].indexOf(c) >= 0), nameCol = H.findIndex(c => c === '종목명' || c === 'name');
  const fx = {KRW: 1}, items = [];
  rows.slice(hi + 1).forEach(r => {
    const code = String(r[codeCol >= 0 ? codeCol : 0] || '').trim().toUpperCase(), name = String(r[nameCol >= 0 ? nameCol : 1] || '').trim();
    const id = (code || norm_(name).toUpperCase()).replace(/[\/\s]/g, '');
    const v = num_(r[pc]), lv = lc >= 0 ? num_(r[lc]) : NaN;
    if (id === 'USDKRW' || id === 'JPYKRW') { const x = lv > 0 ? lv : v; if (x > 0) fx[id.slice(0, 3)] = x; return; }
    items.push({code, n: norm_(name), v, lv});
  });
  let s = 0, k = 0, p = 0, miss = 0;
  (snap.holdings || []).forEach(h => {
    const tk = String(h.t || '').trim().toUpperCase(), nn = norm_(h.n);
    let it = tk ? items.find(x => x.code === tk) : null;
    if (!it) it = items.find(x => x.n === nn) || items.find(x => nn.length >= 2 && x.n.length >= 2 && (x.n.startsWith(nn) || nn.startsWith(x.n)));
    const f = fx[h.c];
    const price = it ? (h.c === 'KRW' ? it.v : it.lv) : NaN;
    if (!(price > 0) || !f) { miss++; return; }
    const val = h.q * price * f;
    s += val;
    if (h.k != null) { const cost = h.kk != null ? h.kk : h.k * f; k += cost; p += val - cost; } // kk: 거래일 환율로 쌓은 원화 원가(있으면 사용)
  });
  let c = 0; (snap.cash || []).forEach(x => { const f = fx[x.c]; if (f) c += x.a * f; });
  const R = Math.round;
  return {t: R(s + c), s: R(s), c: R(c), p: R(p), k: R(k), m: miss};
}
function histSheet_(ss) {
  let sh = ss.getSheetByName(HIST_NAME);
  if (!sh) { sh = ss.insertSheet(HIST_NAME); sh.appendRow(['날짜', '총자산', '주식평가액', '현금', '평가손익', '취득원가', '제외종목수']); sh.getRange('A:A').setNumberFormat('@'); }
  return sh;
}
function recordDaily() {
  const today = Utilities.formatDate(new Date(), 'Asia/Seoul', 'yyyy-MM-dd');
  const dow = +Utilities.formatDate(new Date(), 'Asia/Seoul', 'u'); // 1=월 ... 7=일
  if (dow >= 6) return; // 주말은 건너뜀
  const d = load_().data, snap = d && d.snapshot;
  if (!snap || !snap.holdings || !snap.holdings.length) throw new Error('대시보드가 보유 현황을 아직 올리지 않았습니다. 대시보드의 기기 간 동기화에서 "지금 서버로 저장"을 눌러 주세요.');
  const ss = SpreadsheetApp.openById(SHEET_ID);
  SpreadsheetApp.flush();
  const src = ss.getSheets().filter(x => x.getSheetId() === 0)[0] || ss.getSheets()[0];
  const rec = calc_(snap, src.getDataRange().getDisplayValues());
  const sh = histSheet_(ss), last = sh.getLastRow();
  const row = [today, rec.t, rec.s, rec.c, rec.p, rec.k, rec.m];
  if (last >= 2 && String(sh.getRange(last, 1).getDisplayValue()) === today) sh.getRange(last, 1, 1, 7).setValues([row]);
  else sh.appendRow(row);
}
function readHistory_() {
  const sh = SpreadsheetApp.openById(SHEET_ID).getSheetByName(HIST_NAME), out = {};
  if (!sh || sh.getLastRow() < 2) return out;
  sh.getRange(2, 1, sh.getLastRow() - 1, 7).getDisplayValues().forEach(r => { if (/^\d{4}-\d{2}-\d{2}$/.test(r[0])) out[r[0]] = {t: num_(r[1]), s: num_(r[2]), c: num_(r[3]), p: num_(r[4]), k: num_(r[5]), m: num_(r[6]) || 0}; });
  return out;
}
// 한 번만 실행하세요: 매일 HOUR시(한국시간)에 recordDaily가 돌도록 예약합니다.
function setupTrigger() {
  ScriptApp.getProjectTriggers().forEach(t => { if (t.getHandlerFunction() === 'recordDaily') ScriptApp.deleteTrigger(t); });
  ScriptApp.newTrigger('recordDaily').timeBased().everyDays(1).atHour(HOUR).inTimezone('Asia/Seoul').create();
}

/* ===== 거래내역 탭: 계좌별로 통째로 저장/삭제, 전체 읽기 ===== */
function tradesSheet_(ss) {
  let sh = ss.getSheetByName(TRADES_NAME);
  if (!sh) { sh = ss.insertSheet(TRADES_NAME); sh.appendRow(TRADE_HEAD); sh.setFrozenRows(1); sh.getRange('C:C').setNumberFormat('@'); }
  return sh;
}
function saveTrades_(account, rows) {
  if (!account) return {ok: false, error: '계좌 이름이 없습니다'};
  const lock = LockService.getScriptLock(); lock.waitLock(20000);
  try {
    const ss = SpreadsheetApp.openById(SHEET_ID), sh = tradesSheet_(ss), W = TRADE_HEAD.length;
    const last = sh.getLastRow();
    const keep = last >= 2 ? sh.getRange(2, 1, last - 1, W).getValues().filter(r => String(r[0]) !== String(account)) : [];
    const fresh = (rows || []).map(r => { const x = r.slice(0, W); while (x.length < W) x.push(''); return x; });
    const all = keep.concat(fresh);
    if (last >= 2) sh.getRange(2, 1, last - 1, W).clearContent();
    if (all.length) {
      sh.getRange(2, 3, all.length, 1).setNumberFormat('@');
      sh.getRange(2, 1, all.length, W).setValues(all);
    }
    return {ok: true, saved: fresh.length, total: all.length};
  } finally { lock.releaseLock(); }
}
function readTrades_() {
  const sh = SpreadsheetApp.openById(SHEET_ID).getSheetByName(TRADES_NAME);
  if (!sh || sh.getLastRow() < 2) return [];
  const tz = 'Asia/Seoul';
  return sh.getRange(2, 1, sh.getLastRow() - 1, TRADE_HEAD.length).getValues()
    .filter(r => String(r[0]).trim() && String(r[4]).trim())
    .map(r => r.map((c, i) => (c instanceof Date) ? Utilities.formatDate(c, tz, 'yyyy-MM-dd') : c));
}

/* ===== 환율 탭: 일별 환율(GOOGLEFINANCE)을 시트가 알아서 채우고, 대시보드가 읽어 감 ===== */
// 한 번만 실행: 편집기 위쪽 함수 선택에서 setupFx → 실행
function setupFx() {
  const ss = SpreadsheetApp.openById(SHEET_ID);
  const sh = ss.getSheetByName(FX_NAME) || ss.insertSheet(FX_NAME);
  sh.clear();
  const from = 'DATE(YEAR(TODAY())-1,12,1)'; // 작년 12월 1일부터 오늘까지(연초 기준일 12/31 포함)
  sh.getRange('A1').setFormula('=GOOGLEFINANCE("CURRENCY:USDKRW","price",' + from + ',TODAY(),"DAILY")');
  sh.getRange('D1').setFormula('=GOOGLEFINANCE("CURRENCY:JPYKRW","price",' + from + ',TODAY(),"DAILY")');
  sh.getRange('G1').setValue('A~B열: 달러(USD→KRW) 일별 값, D~E열: 엔(JPY→KRW, 1엔당) 일별 값. 시트가 자동으로 채웁니다. 이 탭은 지우지 마세요.');
}
function readFx_() {
  const ss = SpreadsheetApp.openById(SHEET_ID), sh = ss.getSheetByName(FX_NAME);
  if (!sh) return null;
  const last = sh.getLastRow(), tz = ss.getSpreadsheetTimeZone() || 'Asia/Seoul';
  if (last < 2) return {USD: [], JPY: []};
  const v = sh.getRange(2, 1, last - 1, 5).getValues();
  const pick = ci => v.filter(r => r[ci] instanceof Date && typeof r[ci + 1] === 'number' && r[ci + 1] > 0)
    .map(r => [Utilities.formatDate(r[ci], tz, 'yyyy-MM-dd'), Math.round(r[ci + 1] * 10000) / 10000]);
  return {USD: pick(0), JPY: pick(3)};
}

/* ===== RS 탭: 종목·지수의 일별 종가(GOOGLEFINANCE)를 임시 탭에 계산해서 돌려줌 ===== */
function readSeries_(tickers, from) {
  if (!/^\d{4}-\d{2}-\d{2}$/.test(from)) return {ok: false, error: '시작일 형식 오류'};
  const list = tickers.split(',').map(t => t.trim()).filter(Boolean).slice(0, 4);
  if (!list.length) return {ok: false, error: '티커가 없습니다'};
  const lock = LockService.getScriptLock(); lock.waitLock(25000);
  try {
    const ss = SpreadsheetApp.openById(SHEET_ID), tz = ss.getSpreadsheetTimeZone() || 'Asia/Seoul';
    let sh = ss.getSheetByName(RS_NAME);
    if (!sh) { sh = ss.insertSheet(RS_NAME); try { sh.hideSheet(); } catch (e) {} }
    const p = from.split('-').map(Number), out = {};
    list.forEach(t => {
      if (!/^[A-Za-z0-9.:\-_^]{1,30}$/.test(t)) { out[t] = {ok: false, error: '티커 형식 오류'}; return; }
      sh.clear();
      sh.getRange('A1').setFormula('=GOOGLEFINANCE("' + t + '","close",DATE(' + p[0] + ',' + p[1] + ',' + p[2] + '),TODAY(),"DAILY")');
      SpreadsheetApp.flush();
      let v = [];
      for (let i = 0; i < 12; i++) {
        v = sh.getRange(1, 1, Math.max(sh.getLastRow(), 1), 2).getValues();
        if (v.length > 1 || /^#/.test(String(v[0][0]))) break;
        Utilities.sleep(700); SpreadsheetApp.flush();
      }
      const rows = v.filter(r => r[0] instanceof Date && typeof r[1] === 'number' && r[1] > 0)
        .map(r => [Utilities.formatDate(r[0], tz, 'yyyy-MM-dd'), r[1]]);
      out[t] = rows.length ? {ok: true, rows} : {ok: false, error: '구글 시세에서 ' + t + '의 과거 데이터를 찾지 못했습니다'};
    });
    sh.clear();
    return {ok: true, series: out};
  } finally { lock.releaseLock(); }
}

/* ===== 종목분석 탭: 코스피 신용잔고(날짜, 조원)를 시트 '신용잔고' 탭에 보관 ===== */
function readCredit_() {
  const ss = SpreadsheetApp.openById(SHEET_ID), sh = ss.getSheetByName(CREDIT_NAME);
  if (!sh || sh.getLastRow() < 2) return {ok: false, error: "시트의 '신용잔고' 탭이 비어 있습니다(금투협 자동 수집이 아직 성공하지 못함)"};
  const tz = ss.getSpreadsheetTimeZone() || 'Asia/Seoul', v = sh.getRange(1, 1, sh.getLastRow(), 2).getValues();
  return {ok: true, rows: v.map(r => [r[0] instanceof Date ? Utilities.formatDate(r[0], tz, 'yyyy-MM-dd') : String(r[0]), r[1]])};
}
function saveCredit_(rows) {
  if (!Array.isArray(rows) || !rows.length || rows.length > 5000) return {ok: false, error: '저장할 값이 없거나 너무 많습니다'};
  const lock = LockService.getScriptLock(); lock.waitLock(25000);
  try {
    const ss = SpreadsheetApp.openById(SHEET_ID), tz = ss.getSpreadsheetTimeZone() || 'Asia/Seoul';
    const sh = ss.getSheetByName(CREDIT_NAME) || ss.insertSheet(CREDIT_NAME), map = {};
    if (sh.getLastRow() > 1) sh.getRange(2, 1, sh.getLastRow() - 1, 2).getValues().forEach(r => { const d = r[0] instanceof Date ? Utilities.formatDate(r[0], tz, 'yyyy-MM-dd') : String(r[0]); if (/^\d{4}-\d{2}-\d{2}$/.test(d) && typeof r[1] === 'number') map[d] = r[1]; });
    rows.forEach(r => { if (/^\d{4}-\d{2}-\d{2}$/.test(String(r[0])) && typeof r[1] === 'number' && r[1] > 0) map[r[0]] = r[1]; });
    const days = Object.keys(map).sort(); sh.clear();
    sh.getRange(1, 1, 1, 2).setValues([['날짜', '코스피 신용융자 잔고(조원)']]);
    sh.getRange(2, 1, days.length, 1).setNumberFormat('@'); sh.getRange(2, 1, days.length, 2).setValues(days.map(d => [d, map[d]]));
    return {ok: true, n: days.length};
  } finally { lock.releaseLock(); }
}

/* ===== 코스피 신용잔고 자동 수집: 금융투자협회 FreeSIS "신용공여 잔고 추이"(STATSCU0100000070) =====
 * 화면 뒤에서 쓰는 조회 주소에 날짜 범위를 보내면 일별 값이 옵니다(단위 백만원).
 * 열: TMPV1 날짜, TMPV2 신용거래융자 전체, TMPV3 유가증권(=코스피), TMPV4 코스닥, TMPV5~7 대주, TMPV8 청약자금, TMPV9 예탁증권담보융자
 * 대시보드가 ?credit=1&refresh=1 로 부르거나, setupCreditTrigger를 한 번 실행하면 매일 저녁 자동으로 시트에 쌓입니다. */
const KOFIA_URL = 'https://freesis.kofia.or.kr/meta/getMetaDataList.do';
function fetchKofiaCredit_(fromYmd, toYmd) { // 'yyyyMMdd' 두 개 → [['yyyy-MM-dd', 조원], ...]
  const res = UrlFetchApp.fetch(KOFIA_URL, {
    method: 'post', contentType: 'application/json', muteHttpExceptions: true,
    payload: JSON.stringify({dmSearch: {OBJ_NM: 'STATSCU0100000070BO', tmpV1: 'D', tmpV45: fromYmd, tmpV46: toYmd}})
  });
  const code = res.getResponseCode(), txt = res.getContentText();
  if (code !== 200) throw new Error('금투협 응답 ' + code + ' ' + txt.slice(0, 120));
  let j; try { j = JSON.parse(txt); } catch (e) { throw new Error('금투협 응답이 JSON이 아님: ' + txt.slice(0, 120)); }
  const list = (j && j.ds1) || [];
  return list.filter(r => /^\d{8}$/.test(String(r.TMPV1)) && typeof r.TMPV3 === 'number' && r.TMPV3 > 0)
    .map(r => { const d = String(r.TMPV1); return [d.slice(0, 4) + '-' + d.slice(4, 6) + '-' + d.slice(6, 8), Math.round(r.TMPV3 / 1e4) / 100]; });
}
function updateCredit_() {
  const tz = 'Asia/Seoul', ymd = t => Utilities.formatDate(new Date(t), tz, 'yyyyMMdd'), now = Date.now();
  const sh = SpreadsheetApp.openById(SHEET_ID).getSheetByName(CREDIT_NAME);
  const have = sh && sh.getLastRow() > 1;
  // 금투협은 짧은 시간에 여러 번 요청하면 잠시 막으므로 요청은 한 번만: 시트가 비어 있으면 약 13개월, 있으면 최근 3주(정정 반영)
  const rows = fetchKofiaCredit_(ymd(now - (have ? 21 : 400) * 864e5), ymd(now));
  if (!rows.length) return {ok: false, error: '금투협에서 받은 값이 없습니다(응답은 왔지만 자료가 비어 있음)'};
  const r = saveCredit_(rows);
  return {ok: r.ok, fetched: rows.length, latest: rows.map(x => x[0]).sort().pop(), error: r.error};
}
// 매일 자동 실행용(트리거). 직접 실행해 시험해 볼 수도 있습니다.
function updateCredit() {
  const r = updateCredit_(); Logger.log(JSON.stringify(r));
  if (!r.ok) throw new Error('신용잔고 수집 실패: ' + r.error);
  CacheService.getScriptCache().put('kofiaCredit', JSON.stringify(r), 3 * 3600);
}
// 한 번만 실행: 매일 저녁 7시(한국시간)에 updateCredit이 돌도록 예약합니다(신용잔고는 보통 다음 영업일에 발표).
function setupCreditTrigger() {
  ScriptApp.getProjectTriggers().forEach(t => { if (t.getHandlerFunction() === 'updateCredit') ScriptApp.deleteTrigger(t); });
  ScriptApp.newTrigger('updateCredit').timeBased().everyDays(1).atHour(19).inTimezone('Asia/Seoul').create();
}
function refreshCredit_() { // 대시보드 요청: 3시간 안에 이미 받았으면 금투협은 건너뛰고 시트만 읽음
  const cache = CacheService.getScriptCache();
  const last = cache.get('kofiaCredit');
  let kofia = last ? Object.assign(JSON.parse(last), {skipped: true}) : null; // 최근 결과(성공·실패)를 그대로 알려 줌
  if (!kofia) {
    try { kofia = updateCredit_(); } catch (err) { kofia = {ok: false, error: String(err && err.message || err)}; }
    cache.put('kofiaCredit', JSON.stringify(kofia), kofia.ok ? 3 * 3600 : 30 * 60); // 실패해도 30분은 다시 묻지 않음(차단 방지)
  }
  const out = readCredit_(); out.kofia = kofia; return out;
}

/* ===== 종목분석 탭 '관련 종목 비교 → 자동 추천': 기준 종목과 관련 있는 종목 3개를 Claude API로 고른다 =====
 * 준비(한 번만): Apps Script 편집기 왼쪽 톱니바퀴(프로젝트 설정) > 스크립트 속성 > 속성 추가
 *   이름 ANTHROPIC_API_KEY / 값 본인의 Anthropic API 키.  (키는 코드에 직접 적지 마세요)
 * 새 버전으로 배포하면 외부 요청 권한 승인 창이 한 번 뜹니다. 같은 종목은 6시간 동안 결과를 재사용합니다. */
function readPeers_(ticker) {
  ticker = String(ticker || '').trim();
  if (!/^[A-Za-z0-9.:\-_^]{1,30}$/.test(ticker)) return {ok: false, error: '티커 형식 오류'};
  const key = PropertiesService.getScriptProperties().getProperty('ANTHROPIC_API_KEY');
  if (!key) return {ok: false, error: 'Apps Script 스크립트 속성에 ANTHROPIC_API_KEY가 없습니다'};
  const cache = CacheService.getScriptCache(), ck = 'peers_' + ticker.toUpperCase(), hit = cache.get(ck);
  if (hit) { try { return JSON.parse(hit); } catch (e) {} }
  const prompt = '한국 개인투자자가 종목끼리 수익률을 비교하려고 합니다.\n기준 종목(구글 파이낸스 티커): ' + ticker + '\n'
    + '이 종목과 사업 내용·업종이 가장 비슷한 상장 종목 3개를 고르세요. 거래가 활발한 종목을 우선하고, 기준 종목이 아닌 다른 종목이어야 합니다. '
    + '한국 상장 종목은 실제로 비교할 만한 곳이 있을 때만 넣으세요.\n'
    + '티커 형식(구글 파이낸스): 미국은 VLO처럼 티커만, 한국은 KRX:010950, 일본은 TYO:7203.\n'
    + '설명 없이 JSON만 답하세요: {"subject":{"t":"기준 티커","name":"한국어 종목명"},"peers":[{"t":"티커","name":"한국어 종목명","why":"비슷한 이유 한 줄(한국어)"}]}';
  const res = UrlFetchApp.fetch('https://api.anthropic.com/v1/messages', {
    method: 'post', contentType: 'application/json', muteHttpExceptions: true,
    headers: {'x-api-key': key, 'anthropic-version': '2023-06-01'},
    payload: JSON.stringify({model: AI_MODEL, max_tokens: 700, messages: [{role: 'user', content: prompt}]})
  });
  const code = res.getResponseCode(), body = res.getContentText();
  if (code !== 200) { let m = ''; try { m = JSON.parse(body).error.message; } catch (e) {} return {ok: false, error: 'Claude API 오류 ' + code + (m ? ': ' + m : '')}; }
  let txt = ''; try { txt = JSON.parse(body).content.map(c => c.text || '').join(''); } catch (e) { return {ok: false, error: 'Claude 응답을 읽지 못했습니다'}; }
  const m = txt.match(/\{[\s\S]*\}/); if (!m) return {ok: false, error: 'Claude 응답에 JSON이 없습니다'};
  let j; try { j = JSON.parse(m[0]); } catch (e) { return {ok: false, error: 'Claude 응답 JSON 오류'}; }
  const okT = t => /^[A-Za-z0-9.:\-_^]{1,30}$/.test(String(t || ''));
  const peers = (j.peers || []).filter(p => p && okT(p.t) && String(p.t).toUpperCase() !== ticker.toUpperCase()).slice(0, 3)
    .map(p => ({t: String(p.t).toUpperCase(), name: String(p.name || p.t).slice(0, 40), why: String(p.why || '').slice(0, 120)}));
  if (!peers.length) return {ok: false, error: 'Claude가 관련 종목을 고르지 못했습니다'};
  const out = {ok: true, peers: peers, subject: j.subject && okT(j.subject.t) ? {t: String(j.subject.t), name: String(j.subject.name || '').slice(0, 40)} : null, model: AI_MODEL};
  cache.put(ck, JSON.stringify(out), 21600);
  return out;
}
