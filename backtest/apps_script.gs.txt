/**
 * 백테스트용 Apps Script (주식 대시보드 Apps Script와 별개의 새 프로젝트)
 * 나무증권 Open API(NH PLUG)에서 국내 종목 목록, 일봉, 투자자별 순매수를, DART에서 분기 재무를 받아 옵니다. 조회 전용 — 주문 API는 쓰지 않습니다.
 *
 * 준비(한 번만)
 * 1) script.google.com → 새 프로젝트 → 이 파일 내용을 전부 붙여넣고 저장
 * 2) 아래 SECRET_KEY를 나만 아는 긴 문자열로 바꾸기(백테스트 화면에 같은 값을 입력)
 * 3) 왼쪽 톱니바퀴(프로젝트 설정) > 스크립트 속성 > 속성 추가
 *    이름 NH_APP_KEY / 값 nhplug.com 앱키,  이름 NH_APP_SECRET / 값 앱시크릿  (키는 코드에 직접 적지 마세요)
 *    (재무 기능) 이름 DART_API_KEY / 값 opendart.fss.or.kr 인증키 40자리
 * 4) 배포 > 새 배포 > 유형: 웹 앱 > 실행 계정: 나 > 액세스 권한: 모든 사용자 → 나온 웹 앱 주소를 백테스트 화면에 입력
 *    코드를 고친 뒤에는 "배포 관리 → 수정(연필) → 버전: 새 버전"으로 배포(주소가 그대로 유지됨)
 * 확인용: 편집기 위쪽 함수 선택에서 testBt → 실행 → 실행 로그 확인
 */
const SECRET_KEY = '여기에-나만-아는-긴-비밀번호를-적으세요';

function out_(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}
// ?universe=1 : 국내 종목 목록. 한 종목 = [코드, 이름, 시장(1 코스피·4 코스닥), 코스피200(1/0), 코스닥150(1/0), 거래정지(1/0)]
// ?candles=005930:1,035720:1&n=1400 : 종목별 일봉(오래된 날짜 → 최근). 한 요청에 최대 30종목.
//   한 봉 = [yyyyMMdd, 시가, 고가, 저가, 종가, 거래량, 기준가, 락구분코드] — 분할·배당락 보정은 백테스트 화면에서 기준가로 함
// ?dart=005930&q=8 : DART 분기별 재무(최근 q개 보고서) + 주식 수. 금액은 원 단위
// ?flows=005930:1,…&n=999 : 종목별 투자자 순매수(오래된 날짜 → 최근, 최대 999일). 한 줄 = [yyyyMMdd, 외국인, 기관계, 개인] 순매수 수량(주)
function doGet(e) {
  const q = (e && e.parameter) || {};
  if (q.key !== SECRET_KEY) return out_({ok: false, error: '비밀번호가 맞지 않습니다'});
  try {
    if (q.universe) return out_({ok: true, list: universe_()});
    if (q.candles) return out_(candlesMany_(String(q.candles), +q.n || 300));
    if (q.flows) return out_(flowsMany_(String(q.flows), +q.n || 300));
    if (q.dart) return out_(dartFin_(String(q.dart), Math.max(1, Math.min(+q.q || 8, 12))));
    return out_({ok: true, hello: '백테스트 Apps Script 연결됨'});
  } catch (err) { return out_({ok: false, error: String(err && err.message || err)}); }
}

/* ===== 나무 토큰·호출 =====
 * 토큰(하루짜리 입장권)은 24시간 유효합니다. 발급할 때마다 증권사 보안 알림이 가므로
 * 스크립트 속성(NH_TOKEN, NH_TOKEN_EXP)에 보관해 재사용하고, 만료됐거나 401(토큰 무효)일 때만 새로 받습니다. */
const NH_BASE = 'https://api.nhplug.com:8443';
const NH_PATHS = ['/krstock/quote/v1/period', '/krstock/quote/v1/currentInvestor']; // 시세 조회만 허용(주문 주소는 넣지 않음)
function nhToken_(force, bad) { // bad: 401을 받은 토큰(이것만 버리고 새로 받음)
  const p = PropertiesService.getScriptProperties();
  const cur = () => { const t = p.getProperty('NH_TOKEN'), x = +p.getProperty('NH_TOKEN_EXP') || 0; return t && Date.now() < x && t !== bad ? t : ''; };
  if (!force && cur()) return cur();
  const lock = LockService.getScriptLock(); lock.waitLock(20000);
  try {
    if (cur()) return cur(); // 기다리는 동안 다른 요청이 이미 새로 받았으면 그걸 씀
    const key = p.getProperty('NH_APP_KEY'), sec = p.getProperty('NH_APP_SECRET');
    if (!key || !sec) throw new Error('Apps Script 스크립트 속성에 NH_APP_KEY / NH_APP_SECRET이 없습니다');
    const q = 'appkey=' + encodeURIComponent(key) + '&appsecretkey=' + encodeURIComponent(sec) + '&grant_type=client_credentials&scope=oob';
    const res = UrlFetchApp.fetch(NH_BASE + '/oauth2/token?' + q, {method: 'post', contentType: 'application/x-www-form-urlencoded', muteHttpExceptions: true});
    const code = res.getResponseCode(); let j = {}; try { j = JSON.parse(res.getContentText()); } catch (e) {}
    if (code !== 200 || !j.access_token) throw new Error('나무 토큰 발급 실패(' + code + ') ' + String(j.error_description || j.rsp_msg || j.error || '').slice(0, 120));
    // 만료 10분 전부터는 새로 받도록 여유를 둠
    p.setProperties({NH_TOKEN: j.access_token, NH_TOKEN_EXP: String(Date.now() + ((+j.expires_in || 86400) - 600) * 1000)});
    return j.access_token;
  } finally { lock.releaseLock(); }
}
function nhCall_(path, input) {
  if (NH_PATHS.indexOf(path) < 0) throw new Error('허용되지 않은 나무 API: ' + path);
  const go = tok => UrlFetchApp.fetch(NH_BASE + path, {method: 'post', contentType: 'application/json', headers: {Authorization: 'Bearer ' + tok}, payload: JSON.stringify({Input_0: input || {}}), muteHttpExceptions: true});
  let tok = nhToken_(), res = go(tok);
  if (res.getResponseCode() === 401) res = go(nhToken_(true, tok));
  const code = res.getResponseCode(), txt = res.getContentText();
  let j; try { j = JSON.parse(txt); } catch (e) { throw new Error('나무 응답(' + code + ')이 JSON이 아님: ' + txt.slice(0, 120)); }
  const m = j.message || {}, cd = String(j.rsp_cd || m.code || m.msg_code || j.error_code || ''), msg = String(j.rsp_msg || m.msg || m.usr_msg || j.error_description || '');
  if (code === 429) throw new Error('나무 API 요청이 너무 잦습니다(429). 잠시 뒤 다시 시도하세요');
  if (code !== 200) throw new Error('나무 응답 ' + code + ' ' + cd + ' ' + msg);
  return {j, cd, msg};
}

/* ===== 종목 목록: 나무 종목 마스터(m_new_stock.mst, 인증 불필요, CP949 고정 길이 237바이트) ===== */
function universe_() {
  const cache = CacheService.getScriptCache(), n = +cache.get('univ_n') || 0;
  if (n) {
    const parts = cache.getAll(Array.from({length: n}, (_, i) => 'univ_' + i));
    if (Object.keys(parts).length === n) return JSON.parse(Array.from({length: n}, (_, i) => parts['univ_' + i]).join(''));
  }
  const res = UrlFetchApp.fetch('https://www.nhplug.com/instruments/m_new_stock.mst', {muteHttpExceptions: true});
  if (res.getResponseCode() !== 200) throw new Error('나무 종목 목록을 받지 못했습니다(' + res.getResponseCode() + ')');
  const list = parseStockMst_(res.getContent());
  try { // 캐시 한 칸이 100KB 제한이라 나눠서 6시간 보관
    const s = JSON.stringify(list), size = 90000, put = {};
    let i = 0; for (; i * size < s.length; i++) put['univ_' + i] = s.substr(i * size, size);
    put.univ_n = String(i); cache.putAll(put, 21600);
  } catch (e) {}
  return list;
}
function parseStockMst_(b) {
  const R = 237, out = [], txt = (i, n) => String.fromCharCode.apply(null, b.slice(i, i + n).map(x => x & 255)).replace(/ +$/, '');
  if (b.length % R) throw new Error('종목 목록 형식이 바뀌었습니다(크기 ' + b.length + ')');
  for (let i = 0; i < b.length; i += R) {
    const code = txt(i, 6), mkt = txt(i + 6, 1);
    if (mkt !== '1' && mkt !== '4') continue; // ETN 등 제외
    let nb = b.slice(i + 7, i + 48);
    if (nb[0] === 42 || nb[0] === 35) nb = nb.slice(1); // 이름 앞 표시 * 코스피200, # 코스닥150
    const name = Utilities.newBlob(nb).getDataAsString('MS949').replace(/[\s\u0000]+$/, '');
    const k200 = txt(i + 168, 1); // 코스피200 산업군(0이면 미편입)
    out.push([code, name, mkt, k200 && k200 !== '0' ? 1 : 0, txt(i + 172, 1) === 'Y' ? 1 : 0, txt(i + 161, 1) === 'Y' ? 1 : 0]);
  }
  return out;
}

/* ===== 일봉: 국내주식 기간별 시세(/krstock/quote/v1/period, 주기 1=일) ===== */
function candles_(code, mkt, n) {
  const map = {}; let edate = Utilities.formatDate(new Date(), 'Asia/Seoul', 'yyyyMMdd');
  for (let page = 0; page < 8; page++) {
    const want = Math.min(n - Object.keys(map).length, 2000);
    if (want <= 0) break;
    const r = nhCall_('/krstock/quote/v1/period', {market_cd: 'KRX', iem_cd: code, mrkt_div_cls_code: mkt, edate, array_cnt: String(want), maxavg: '0', gubun: '1', xtick: '0', today_cls_code: '0', fake_tick: '0', sur_flag: '0', sur_gb_day_cnt: '00', sur_bf_end_time: '000000', out1_scale_change: '0', out2_scale_change: '0', view_main_yn: 'Y'});
    const rows = r.j.Output_1 || [];
    let oldest = '', added = 0;
    rows.forEach(x => {
      const d = String(x.bsop_date || ''); if (!/^\d{8}$/.test(d)) return;
      if (!map[d]) added++;
      map[d] = [d, +x.stck_oprc || 0, +x.stck_hgpr || 0, +x.stck_lwpr || 0, +x.stck_prpr || 0, +x.vol || 0, +x.stck_sdpr || 0, String(x.flng_cls_code || '0').trim()];
      if (!oldest || d < oldest) oldest = d;
    });
    if (!added || rows.length < want) break; // 더 오래된 데이터가 없음
    const od = new Date(+oldest.slice(0, 4), +oldest.slice(4, 6) - 1, +oldest.slice(6) - 1); // 가장 오래된 날 하루 전부터 이어서
    edate = Utilities.formatDate(od, 'Asia/Seoul', 'yyyyMMdd');
    Utilities.sleep(220); // 초당 5회 제한
  }
  return Object.keys(map).sort().map(d => map[d]);
}
function candlesMany_(codes, n) { return many_(codes, Math.max(5, Math.min(n, 3000)), candles_); }
function many_(codes, n, fn) { // 한 요청에 최대 30종목, 종목 사이 220ms 쉼(초당 5회 제한)
  const list = codes.split(',').map(s => s.trim()).filter(s => /^[0-9A-Z]{6}:[14]$/.test(s)).slice(0, 30);
  if (!list.length) return {ok: false, error: '종목코드가 없습니다'};
  const data = {}, errors = {};
  list.forEach((s, i) => {
    const [code, mkt] = s.split(':');
    if (i) Utilities.sleep(220);
    try { data[code] = fn(code, mkt, n); } catch (err) { errors[code] = String(err && err.message || err); }
  });
  return {ok: true, data, errors};
}

/* ===== 투자자별 순매수: 주식현재가 투자자(/krstock/quote/v1/currentInvestor) =====
 * 최근 날짜부터 array_cnt(최대 999)일. 날짜를 지정해 더 옛날을 이어 받는 입력은 없음 */
function flows_(code, mkt, n) {
  const r = nhCall_('/krstock/quote/v1/currentInvestor', {market_cd: 'KRX', iem_cd: code, array_cnt: String(Math.max(1, Math.min(n, 999)))});
  const map = {};
  (r.j.Output_0 || []).forEach(x => {
    const d = String(x.bsop_date1 || ''); if (!/^\d{8}$/.test(d)) return;
    map[d] = [d, +x.frgn_ntby_qty || 0, +x.gigwan || 0, +x.person || 0];
  });
  return Object.keys(map).sort().map(d => map[d]);
}
function flowsMany_(codes, n) { return many_(codes, n, flows_); }

/* ===== DART(금감원 전자공시 오픈API) 재무 =====
 * 종목코드(6자리) → DART 고유번호(8자리)는 corpCode.xml(zip)에서 찾아 6시간 캐시.
 * 분기 재무는 fnlttSinglAcntAll(전체 재무제표, 연결 CFS 없으면 별도 OFS), 주식 수는 stockTotqySttus.
 * 손익(매출·영업이익·순이익)은 {a: 이번 보고서 금액(분기 3개월), c: 누적}으로 줌 — 사업보고서(4분기)의 a는 1년 전체라 화면에서 3분기 누적을 빼서 4분기만 구함 */
const DART_BASE = 'https://opendart.fss.or.kr/api/';
function dartKey_() {
  const key = PropertiesService.getScriptProperties().getProperty('DART_API_KEY');
  if (!key) throw new Error('Apps Script 스크립트 속성에 DART_API_KEY가 없습니다');
  return key;
}
function dartGet_(api, params) {
  const q = Object.keys(params).map(k => k + '=' + encodeURIComponent(params[k])).join('&');
  const res = UrlFetchApp.fetch(DART_BASE + api + '?crtfc_key=' + encodeURIComponent(dartKey_()) + '&' + q, {muteHttpExceptions: true});
  let j; try { j = JSON.parse(res.getContentText()); } catch (e) { throw new Error('DART 응답(' + res.getResponseCode() + ')을 읽지 못했습니다'); }
  if (j.status === '013') return []; // 조회된 데이터 없음
  if (j.status !== '000') throw new Error('DART ' + j.status + ' ' + (j.message || ''));
  return j.list || [];
}
// 캐시 한 칸이 100KB 제한이라 큰 값은 나눠서 보관
function cacheGetJson_(name) {
  const cache = CacheService.getScriptCache(), n = +cache.get(name + '_n') || 0;
  if (!n) return null;
  const parts = cache.getAll(Array.from({length: n}, (_, i) => name + '_' + i));
  return Object.keys(parts).length === n ? JSON.parse(Array.from({length: n}, (_, i) => parts[name + '_' + i]).join('')) : null;
}
function cachePutJson_(name, obj, sec) {
  try {
    const s = JSON.stringify(obj), size = 90000, put = {};
    let i = 0; for (; i * size < s.length; i++) put[name + '_' + i] = s.substr(i * size, size);
    put[name + '_n'] = String(i); CacheService.getScriptCache().putAll(put, sec);
  } catch (e) {}
}
function corpMap_() { // {종목코드: [고유번호, 회사명]} 상장사만
  const hit = cacheGetJson_('corp'); if (hit) return hit;
  const res = UrlFetchApp.fetch(DART_BASE + 'corpCode.xml?crtfc_key=' + encodeURIComponent(dartKey_()), {muteHttpExceptions: true});
  if (res.getResponseCode() !== 200) throw new Error('DART 회사 목록을 받지 못했습니다(' + res.getResponseCode() + ')');
  let files;
  try { files = Utilities.unzip(res.getBlob().setContentType('application/zip')); }
  catch (e) { throw new Error('DART 회사 목록 압축을 풀지 못했습니다(인증키 확인): ' + res.getContentText().slice(0, 120)); }
  const xml = files[0].getDataAsString('UTF-8'), map = {};
  xml.split('</list>').forEach(part => { // 한 회사 = <list> … </list>. 상장사만 stock_code가 있음
    const sc = /<stock_code>\s*([0-9A-Z]{6})\s*<\/stock_code>/.exec(part); if (!sc) return;
    const cc = /<corp_code>(\d{8})<\/corp_code>/.exec(part), nm = /<corp_name>([^<]*)<\/corp_name>/.exec(part);
    if (cc) map[sc[1]] = [cc[1], nm ? nm[1].trim() : ''];
  });
  cachePutJson_('corp', map, 21600);
  return map;
}
const DART_RPT = {'11013': 1, '11012': 2, '11014': 3, '11011': 4}; // 보고서 코드 → 분기(1분기·반기·3분기·사업)
function dartNum_(v) { const s = String(v == null ? '' : v).replace(/,/g, '').trim(); return /^-?\d+(\.\d+)?$/.test(s) ? +s : null; }
function dartPeriod_(corp, y, r) {
  const ck = 'df_' + corp + '_' + y + r, cache = CacheService.getScriptCache(), hit = cache.get(ck);
  if (hit) return JSON.parse(hit);
  let fs = 'CFS', list = dartGet_('fnlttSinglAcntAll.json', {corp_code: corp, bsns_year: y, reprt_code: r, fs_div: 'CFS'});
  if (!list.length) { fs = 'OFS'; list = dartGet_('fnlttSinglAcntAll.json', {corp_code: corp, bsns_year: y, reprt_code: r, fs_div: 'OFS'}); }
  if (!list.length) { cache.put(ck, 'null', 3600); return null; }
  const norm = s => String(s || '').replace(/[\s·ㆍ()（）]/g, '');
  const find = (sjs, ids, names) => { // 계정 id로 먼저, 없으면 계정 이름으로
    for (const sj of sjs) for (const x of list) if (x.sj_div === sj && ids.indexOf(x.account_id) >= 0) return x;
    for (const sj of sjs) for (const x of list) if (x.sj_div === sj && names.indexOf(norm(x.account_nm)) >= 0) return x;
    return null;
  };
  const bs = (ids, names) => { const x = find(['BS'], ids, names); return x ? dartNum_(x.thstrm_amount) : null; };
  const is = (ids, names) => { const x = find(['IS', 'CIS'], ids, names); return x ? {a: dartNum_(x.thstrm_amount), c: dartNum_(x.thstrm_add_amount)} : null; };
  const out = {y: +y, q: DART_RPT[r], fs,
    eq: bs(['ifrs-full_Equity'], ['자본총계']),
    eqo: bs(['ifrs-full_EquityAttributableToOwnersOfParent'], ['지배기업의소유주에게귀속되는자본', '지배기업소유주지분', '지배기업의소유지분']),
    cash: bs(['ifrs-full_CashAndCashEquivalents'], ['현금및현금성자산']),
    assets: bs(['ifrs-full_Assets'], ['자산총계']),
    liab: bs(['ifrs-full_Liabilities'], ['부채총계']),
    rev: is(['ifrs-full_Revenue'], ['매출액', '수익매출액', '영업수익', '매출']),
    op: is(['dart_OperatingIncomeLoss'], ['영업이익', '영업이익손실']),
    ni: is(['ifrs-full_ProfitLoss'], ['당기순이익', '당기순이익손실', '분기순이익', '반기순이익']),
    nio: is(['ifrs-full_ProfitLossAttributableToOwnersOfParent'], ['지배기업의소유주에게귀속되는당기순이익', '지배기업소유주지분순이익'])};
  try { // 주식 수(발행 − 자기주식). 합계 줄이 없으면 보통주+우선주
    const st = dartGet_('stockTotqySttus.json', {corp_code: corp, bsns_year: y, reprt_code: r});
    const row = st.find(x => /합계/.test(x.se));
    const sum = k => row ? dartNum_(row[k]) : st.filter(x => /보통|우선/.test(x.se)).reduce((t, x) => t + (dartNum_(x[k]) || 0), 0);
    out.shares = sum('istc_totqy'); out.treasury = sum('tesstk_co');
  } catch (e) {}
  cache.put(ck, JSON.stringify(out), 21600);
  return out;
}
function dartFin_(code, n) {
  if (!/^[0-9A-Z]{6}$/.test(code)) return {ok: false, error: '종목코드 6자리를 넣어 주세요'};
  const hit = corpMap_()[code];
  if (!hit) return {ok: false, error: 'DART에서 이 종목을 찾지 못했습니다(' + code + ')'};
  const corp = hit[0], y0 = +Utilities.formatDate(new Date(), 'Asia/Seoul', 'yyyy'), out = [];
  let tried = 0;
  for (let y = y0; y >= y0 - 4 && out.length < n; y--) {
    for (const r of ['11011', '11014', '11012', '11013']) { // 최근 보고서부터: 사업(4Q)·3Q·반기·1Q
      if (out.length >= n) break;
      if (y === y0 && r === '11011') continue; // 올해 사업보고서는 아직 없음
      const p = dartPeriod_(corp, String(y), r); tried++;
      if (p) out.push(p);
      else if (!out.length && tried >= 6) return {ok: true, code, corp, name: hit[1], periods: []}; // 최근 1년 반 동안 아무것도 없음
    }
  }
  return {ok: true, code, corp, name: hit[1], periods: out};
}
